#+begin_src scheme
(import :yuanxu/object-syntax/path-slots
        (only-in :clan/poo/object .get .ref))

;; Domain-free design probe, not a product selector/admission implementation.
(def (selected-values receiver names)
  (map (lambda (name) (.ref receiver name)) names))

(def Profile
  (.o (:: effective-profile [])
    build: (.o value: 'parent-build)
    test: (.o value: 'parent-test)
    package: (.o value: 'parent-package)
    pipeline:
    (.o pipeline-name: 'default
        selected-checks: '(build test package)
        checks: (selected-values effective-profile selected-checks))))

(.def Project funflow: (.o profiles: (.o gerbil-scheme: Profile)))
(def parent-checks (.get Project funflow profiles gerbil-scheme pipeline checks))

(.def (Release @ Project)
  funflow.profiles.gerbil-scheme.build: =>.+ {value: 'release-build}
  funflow.profiles.gerbil-scheme.test: =>.+ {value: 'release-test}
  funflow.profiles.gerbil-scheme.package: =>.+ {value: 'release-package}
  funflow.profiles.gerbil-scheme.pipeline: =>.+ {pipeline-name: 'release})

(.def (Extended @ Release)
  funflow.profiles.gerbil-scheme: =>.+
  {smoke: (.o value: 'smoke)
   pipeline: =>.+ {selected-checks: '(build test package smoke)}})

(def assertions 0)
(def (assert! label condition)
  (unless condition (error "design receiver probe failed" label))
  (set! assertions (+ assertions 1)))
(def (values-of checks) (map (lambda (check) (.get check value)) checks))
(def release-checks (.get Release funflow profiles gerbil-scheme pipeline checks))
(def extended-checks (.get Extended funflow profiles gerbil-scheme pipeline checks))
(assert! 'parent-cached (equal? (values-of parent-checks) '(parent-build parent-test parent-package)))
(assert! 'four-paths-retained
  (and (equal? (values-of release-checks) '(release-build release-test release-package))
       (eq? (.get Release funflow profiles gerbil-scheme pipeline pipeline-name) 'release)))
(assert! 'dynamic-child-selection
  (equal? (values-of extended-checks) '(release-build release-test release-package smoke)))
(for-each
  (lambda (check name)
    (assert! name (eq? check (.ref (.get Extended funflow profiles gerbil-scheme) name))))
  extended-checks '(build test package smoke))
(assert! 'warm-list (eq? extended-checks (.get Extended funflow profiles gerbil-scheme pipeline checks)))
(assert! 'parent-unchanged (eq? parent-checks (.get Project funflow profiles gerbil-scheme pipeline checks)))
(.def (Frozen @ Extended)
  funflow.profiles.gerbil-scheme.pipeline: =>.+ {checks: parent-checks})
(assert! 'frozen-parent-negative
  (equal? (values-of (.get Frozen funflow profiles gerbil-scheme pipeline checks))
          '(parent-build parent-test parent-package)))
(displayln "PASS: native lexical receiver / selected names design probe assertions=" assertions)
#+end_src
