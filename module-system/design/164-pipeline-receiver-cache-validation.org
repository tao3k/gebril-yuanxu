* Test variables versus the shared module object abstraction

parent-checks and child-checks hold observations for assertions; they do not
assemble or repair a configured Pipeline. Maintained Agentic AI Lab and User Interface
Modules share the same native object/scope semantics. The Pipeline's computed
checks slot owns list construction from the effective Profile, and the selected
plan adapter consumes it internally. Ordinary users only refine target paths.
The isolated probe validates that native relationship; it does not establish
the unimplemented common Module scope, path syntax or Loader integration.

* Complete executed native probe

#+name: native-pipeline-receiver-full-probe
#+begin_src scheme
(import "/tmp/agenticailab-pipeline-receiver-pin/object")

(def assertions 0)
(def (assert-equal label actual expected)
  (set! assertions (+ assertions 1))
  (unless (equal? actual expected)
    (error "assertion failed" label actual expected)))
(def (assert-true label actual) (assert-equal label (if actual #t #f) #t))
(def pipeline-builds 0)
(def checks-builds 0)

(.def (Profile @ [])
  build: (.o check-name: 'build value: 'parent-build)
  test: (.o check-name: 'test value: 'parent-test)
  package: (.o check-name: 'package value: 'parent-package)
  pipeline:
  (let ((profile @))
    (set! pipeline-builds (+ pipeline-builds 1))
    (.o pipeline-name: 'default
        checks:
        (begin
          (set! checks-builds (+ checks-builds 1))
          (list (.get profile build)
                (.get profile test)
                (.get profile package))))))

(.def ParentScenario
  funflow: (.o profiles: (.o gerbil-scheme: Profile)))
(def (scenario-profile root) (.get root funflow profiles gerbil-scheme))
(def (scenario-pipeline root) (.get root funflow profiles gerbil-scheme pipeline))
(def (scenario-checks root) (.get root funflow profiles gerbil-scheme pipeline checks))
(def (values-of checks) (map (lambda (check) (.get check value)) checks))

;; Force the parent Pipeline AND its checks before defining the child.
(def parent-profile (scenario-profile ParentScenario))
(def parent-pipeline (scenario-pipeline ParentScenario))
(assert-equal 'pipeline-demand-count pipeline-builds 1)
(assert-equal 'checks-still-lazy checks-builds 0)
(def parent-checks (scenario-checks ParentScenario))
(assert-equal 'parent-values (values-of parent-checks)
              '(parent-build parent-test parent-package))
(assert-equal 'parent-checks-count checks-builds 1)

;; Native expansion of three shared-prefix path refinements.
(.def (ChildScenario @ ParentScenario)
  funflow: =>.+
  (.o profiles: =>.+
      (.o gerbil-scheme: =>.+
          (.o build: =>.+ (.o value: 'child-build)
              test: =>.+ (.o value: 'child-test)
              package: =>.+ (.o value: 'child-package)))))

(assert-equal 'child-declaration-does-not-force-pipeline pipeline-builds 1)
(assert-equal 'child-declaration-does-not-force-checks checks-builds 1)
(def child-profile (scenario-profile ChildScenario))
(assert-true 'different-profile (not (eq? child-profile parent-profile)))
(def child-pipeline (scenario-pipeline ChildScenario))
(assert-true 'different-pipeline (not (eq? child-pipeline parent-pipeline)))
(assert-equal 'new-pipeline-receiver pipeline-builds 2)
(assert-equal 'child-checks-still-lazy checks-builds 1)
(def child-checks (scenario-checks ChildScenario))
(assert-equal 'child-values (values-of child-checks)
              '(child-build child-test child-package))
(assert-equal 'child-check-order
              (map (lambda (check) (.get check check-name)) child-checks)
              '(build test package))
(assert-equal 'child-check-count checks-builds 2)
(assert-true 'list-has-effective-build
             (eq? (car child-checks) (.get child-profile build)))
(assert-true 'list-has-effective-test
             (eq? (cadr child-checks) (.get child-profile test)))
(assert-true 'list-has-effective-package
             (eq? (caddr child-checks) (.get child-profile package)))
(assert-true 'child-pipeline-cache
             (eq? child-pipeline (scenario-pipeline ChildScenario)))
(assert-true 'child-check-list-cache
             (eq? child-checks (scenario-checks ChildScenario)))
(assert-equal 'warm-pipeline-count pipeline-builds 2)
(assert-equal 'warm-check-count checks-builds 2)
(assert-true 'parent-pipeline-cache-unchanged
             (eq? parent-pipeline (scenario-pipeline ParentScenario)))
(assert-true 'parent-check-list-cache-unchanged
             (eq? parent-checks (scenario-checks ParentScenario)))
(assert-equal 'parent-values-unchanged (values-of (scenario-checks ParentScenario))
              '(parent-build parent-test parent-package))

;; Also refine Pipeline itself, after ChildScenario's Pipeline is cached.
(.def (ReleaseScenario @ ChildScenario)
  funflow: =>.+
  (.o profiles: =>.+
      (.o gerbil-scheme: =>.+
          (.o pipeline: =>.+ (.o pipeline-name: 'release)))))
(assert-equal 'release-declaration-lazy pipeline-builds 2)
(assert-equal 'release-name
              (.get ReleaseScenario funflow profiles gerbil-scheme pipeline pipeline-name)
              'release)
(assert-equal 'release-values (values-of (scenario-checks ReleaseScenario))
              '(child-build child-test child-package))
(assert-equal 'release-receiver-count pipeline-builds 3)
(assert-equal 'release-checks-count checks-builds 3)

;; Sibling isolation after both parent/child have been forced.
(.def (SiblingScenario @ ParentScenario)
  funflow: =>.+
  (.o profiles: =>.+
      (.o gerbil-scheme: =>.+
          (.o build: =>.+ (.o value: 'sibling-build)))))
(assert-equal 'sibling-values (values-of (scenario-checks SiblingScenario))
              '(sibling-build parent-test parent-package))
(assert-equal 'child-values-still-current (values-of (scenario-checks ChildScenario))
              '(child-build child-test child-package))

;; Negative control: an explicitly frozen parent list really IS stale.
(.def (FrozenProfile @ Profile)
  build: =>.+ (.o value: 'should-be-new)
  pipeline: (.o checks: parent-checks))
(assert-equal 'frozen-direct-build (.get FrozenProfile build value) 'should-be-new)
(assert-equal 'frozen-negative-control
              (values-of (.get FrozenProfile pipeline checks))
              '(parent-build parent-test parent-package))

(displayln "PASS assertions=" assertions)
(displayln "parent=(parent-build parent-test parent-package)")
(displayln "child=(child-build child-test child-package)")
(displayln "release=(child-build child-test child-package), name=release")
(displayln "sibling=(sibling-build parent-test parent-package)")
